cmake_minimum_required(VERSION 3.20)
project(miniima_exact VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

add_library(miniima_exact STATIC cpp/src/exact_index.cpp)
target_include_directories(miniima_exact PUBLIC cpp/include)
if(MSVC)
  target_compile_options(miniima_exact PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(miniima_exact PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()

option(MINIIMA_BUILD_PYTHON "Build the pybind11 Python extension" ON)
if(MINIIMA_BUILD_PYTHON)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_exact NO_EXTRAS cpp/bindings/module.cpp)
  target_link_libraries(_exact PRIVATE miniima_exact)
  # MinGW is supported as well as MSVC. Bundle its C++ runtime statically so the
  # installed extension can load without MSYS2 being on the user's PATH.
  if(MINGW)
    target_link_options(_exact PRIVATE -static-libgcc -static-libstdc++ -static)
  endif()
  install(TARGETS _exact LIBRARY DESTINATION miniima RUNTIME DESTINATION miniima)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(exact_index_tests cpp/tests/test_exact_index.cpp)
  target_link_libraries(exact_index_tests PRIVATE miniima_exact)
  if(MINGW)
    target_link_options(exact_index_tests PRIVATE -static-libgcc -static-libstdc++ -static)
  endif()
  add_test(NAME exact_index_edges COMMAND exact_index_tests)
endif()
